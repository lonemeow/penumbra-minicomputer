# Per-test compiler flag overrides for tests that don't specify
# `{ dg-options "..." }` in their source but still need specific
# flags to compile or run correctly under our harness.
#
# Format: <pattern>  <flags...>
# Patterns match the same way as excludes.txt (gitignore-style
# suffix match on the test path).  Flags are appended to the
# test.c compile command after our base -std=gnu89.

# Multiple for-init declarations in one function collide under
# gnu89 because clang places extension-accepted for-init vars at
# function scope.  Under gnu99+ each for-statement has its own
# scope and they don't collide.
UnitTests/Float/classify.c          -std=gnu99

# gnu89's integer-constant promotion ladder is {int, long,
# unsigned long} — no long long.  The expected-value arrays
# contain decimal constants like 2810079215 that overflow signed
# long, get typed as unsigned long, and wrap to a positive s64
# when negated.  The C99+ ladder adds long long so the negation
# stays negative.
Regression/C/gcc-c-torture/execute/shiftdi-2.c    -std=gnu99

# llvm-test-suite's CMakeLists.txt groups this under
# TestRequiresFWrapV: the body is full of `x + y` where one side
# may be INT_MIN, and expects two's-complement wrap.  The test
# does not declare -fwrapv itself.
Regression/C/gcc-c-torture/execute/950704-1.c     -fwrapv

# Test is flagged `dg-add-options stack_size` upstream — DejaGnu
# defines STACK_SIZE based on target memory.  Without it, the test
# tries to malloc 2 × 128 KB, overflowing our 128 KB bump heap;
# both pointers come back NULL and the test then scribbles -1 over
# low memory (starting at address 0), trashing .text and hanging
# the ISS until +max-insn.  16384 is the conventional small-target
# value; BLOCK_SIZE becomes 2048, fits comfortably.
Regression/C/gcc-c-torture/execute/960521-1.c     -DSTACK_SIZE=16384

# Test relies on C99's "main falling off end implies return 0".
# Under -ffreestanding, main is not special and clang emits
# `ret i32 undef`; R1 happens to hold val=2 when main returns, so
# the test exits 2 instead of 0.  -fno-builtin preserves our
# opt-out of libc-name folding while -fhosted re-enables the
# main-returns-zero guarantee.
Regression/C/gcc-c-torture/execute/alias-access-path-1.c  -fhosted -fno-builtin
Regression/C/gcc-c-torture/execute/strlen-2.c             -fhosted -fno-builtin
Regression/C/gcc-c-torture/execute/strlen-4.c             -fhosted -fno-builtin
